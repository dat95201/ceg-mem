% ── notation, defined once so the body never hand-writes a symbol ─────────
% Small caps that survive text extraction. \textsc{CEGMem} sets "CEGM" and
% "em" in two different fonts, so pdftotext emits "CEGM EM": the name is
% unsearchable, a reviewer's copy-paste is wrong, and a screen reader spells
% it letter by letter. accsupp attaches the plain spelling as the extractable
% text and moves no glyph. Every small-caps NAME goes through \scname; plain
% \textsc stays available for ordinary small-caps text.
\DeclareRobustCommand{\scname}[1]{%
  \texorpdfstring{\BeginAccSupp{method=escape,ActualText={#1}}\textsc{#1}\EndAccSupp{}}{#1}}

% ── page-limit switch ─────────────────────────────────────────────────────
% A 10+2 submission cannot carry an appendix: the two extra pages are for
% references only. \suppmattertrue makes every reference to appendix content
% resolve to the supplementary archive instead of to a \cref that would dangle.
% The drivers set it; the section bodies never test it directly.
\newif\ifsuppmatter
\suppmatterfalse
\newcommand{\appref}[2]{\ifsuppmatter #2\else\cref{#1}\fi}

% Content that ships only in the full version: floats whose numbers are stated
% in the text anyway, the appendix, and the extended tables. In the submission
% build these vanish and their references resolve through \appref, so the two
% builds share one prose source and cannot drift.
\newcommand{\fullonly}[1]{\ifsuppmatter\else#1\fi}
\newcommand{\supponly}[1]{\ifsuppmatter #1\fi}

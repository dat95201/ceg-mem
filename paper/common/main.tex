% CEGMem - PORTABLE PREPRINT DRIVER

% Builds with a plain TeX Live installation: no acmart, no IEEEtran needed.

% and refs.bib so the two drivers can never drift. See README.md.

\documentclass[10pt,twocolumn]{article}

\usepackage[letterpaper,margin=0.72in,columnsep=0.28in]{geometry}
\usepackage{mathptmx}

% Every font in this driver must be scalable (Type 1 / OpenType): microtype's
% font expansion aborts the build on a bitmap font, and with T1 encoding the
% default cmtt/cmss resolve to bitmap ec* unless lmodern or cm-super is
% installed. Times covers roman and math; these cover the other two families,
% preferring Latin Modern where the installation has it and falling back to
% the psnfss URW fonts, which ship with texlive-fonts-recommended.
\IfFileExists{lmodern.sty}{%
  \renewcommand{\ttdefault}{lmtt}%
  \renewcommand{\sfdefault}{lmss}%
}{%
  \IfFileExists{courier.sty}{\usepackage{courier}}{}%
  \IfFileExists{helvet.sty}{\usepackage[scaled=0.92]{helvet}}{}%
}

\usepackage[numbers,sort&compress]{natbib}
\usepackage{caption}
\usepackage{titlesec}

% Package order below is load-bearing: amsmath (via preamble) before amsthm,
% amsthm before hyperref, hyperref before cleveref, and every \newtheorem
% after cleveref so that cleveref learns the environment names.
\input{preamble}
\usepackage{amsthm}
\usepackage[hidelinks]{hyperref}
\usepackage[nameinlink,capitalise]{cleveref}
\input{preamble-cleveref}

% Compact headings, so pagination on this driver rehearses the submission's.

\titleformat{\subsection}{\normalfont\normalsize\bfseries}{\thesubsection}{0.6em}{}
\titleformat{\subsubsection}{\normalfont\normalsize\itshape}{\thesubsubsection}{0.6em}{}
\titlespacing*{\section}{0pt}{9pt plus 2pt minus 1pt}{4pt}
\titlespacing*{\subsection}{0pt}{7pt plus 1pt minus 1pt}{3pt}
\titlespacing*{\subsubsection}{0pt}{6pt plus 1pt minus 1pt}{2pt}

\captionsetup{font=small,labelfont=bf,skip=4pt}
\setlength{\abovecaptionskip}{5pt}
\setlength{\belowcaptionskip}{2pt}
